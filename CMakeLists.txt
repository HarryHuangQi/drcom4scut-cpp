cmake_minimum_required(VERSION 3.10)
project(drcom4scut_cpp VERSION 0.2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(protocol_tests tests/protocol_tests.cpp)
target_link_libraries(protocol_tests PRIVATE OpenSSL::Crypto)
target_compile_options(protocol_tests PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(drcom4scut main.cpp)
  target_link_libraries(drcom4scut PRIVATE OpenSSL::Crypto)
  target_compile_options(drcom4scut PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS drcom4scut RUNTIME DESTINATION bin)
elseif(WIN32)
  add_executable(drcom4scut windows/main_windows.cpp)
  target_link_libraries(drcom4scut PRIVATE OpenSSL::Crypto ws2_32 iphlpapi crypt32 bcrypt)
  add_executable(drcom4scut-gui WIN32 windows/gui_windows.cpp windows/resources.rc)
  target_link_libraries(drcom4scut-gui PRIVATE ws2_32 iphlpapi shell32 comctl32 ole32 uuid)
  if(MSVC)
    target_compile_options(drcom4scut PRIVATE /W4)
    target_compile_options(drcom4scut-gui PRIVATE /W4)
  else()
    target_compile_options(drcom4scut PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(drcom4scut-gui PRIVATE -Wall -Wextra -Wpedantic -municode)
    target_link_options(drcom4scut-gui PRIVATE -municode)
  endif()
  install(TARGETS drcom4scut drcom4scut-gui RUNTIME DESTINATION bin)
else()
  message(STATUS "Only portable protocol tests are built on this platform")
endif()
enable_testing()
add_test(NAME protocol COMMAND protocol_tests)
